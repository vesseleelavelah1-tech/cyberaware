def check_password_strength():
    print("\n===== PASSWORD STRENGTH CHECKER =====")

    print("IMPORTANT: Enter only a fictional test password.")
    password = input("Enter a test password: ")

    score = 0

    # Check password length
    if len(password) >= 12:
        score += 2
    elif len(password) >= 8:
        score += 1

    # Check uppercase letters
    if any(char.isupper() for char in password):
        score += 1

    # Check lowercase letters
    if any(char.islower() for char in password):
        score += 1

    # Check numbers
    if any(char.isdigit() for char in password):
        score += 1

    # Check special characters
    if any(not char.isalnum() for char in password):
        score += 1

    # Display result
    print("\nPassword Assessment:")

    if score <= 2:
        print("Strength: WEAK")
        print("Advice: Use a longer and more varied password.")

    elif score <= 4:
        print("Strength: MODERATE")
        print("Advice: Add more character variety and length.")

    else:
        print("Strength: STRONG")
        print("Advice: Keep it unique and never reuse it.")

    print("Assessment completed.")

def view_assessment_results():
    import json
    import os

    filename = "cyberaware_assessments.json"

    print("\n===== CYBERAWARE ASSESSMENT RESULTS =====")

    # Check whether the file exists
    if not os.path.exists(filename):
        print("No assessment records found yet.")

    try:
        with open(filename, "r") as file:
            records = json.load(file)

    except json.JSONDecodeError:
        print("Error: Assessment file contains invalid JSON.")

    except OSError:
        print("Error: Unable to read the assessment file.")
        return

    # Check whether records exist
    if not isinstance(records, list) or len(records) == 0:

        return

    total_attempts = len(records)
    total_percentage = 0
    scores = []

    print("\nSAVED STUDENT RESULTS")

    # Display every assessment record
    for number, record in enumerate(records, start=1):

        student_id = record.get("student_id", "Unknown")
        score = record.get("score", 0)
        total = record.get("total_questions", 0)
        percentage = record.get("percentage", 0)

        print("\nAttempt:", number)
        print("Student ID:", student_id)
        print("Score:", score, "/", total)
        print("Percentage:", percentage, "%")

        total_percentage += percentage
        scores.append(percentage)

    # Calculate statistics
    average_score = total_percentage / total_attempts
    highest_score = max(scores)
    lowest_score = min(scores)

    print("\n===== PERFORMANCE SUMMARY =====")
    print("Total assessment attempts:", total_attempts)
    print("Average percentage:", round(average_score, 2), "%")
    print("Highest percentage:", highest_score, "%")
    print("Lowest percentage:", lowest_score, "%")

def student_performance_report():
    import json
    import os

    progress_file = "cyberaware_progress.json"
    assessment_file = "cyberaware_assessments.json"

    print("\n===== CYBERAWARE STUDENT PERFORMANCE REPORT =====")

    # Function to read saved JSON records
    def load_records(filename):
        if not os.path.exists(filename):
            return []

        try:
            with open(filename, "r") as file:
                records = json.load(file)

            if isinstance(records, list):
                return records

            print("Warning: Invalid record format in", filename)
            return []

        except (json.JSONDecodeError, OSError):
            print("Warning: Could not read", filename)
            return []

    progress_records = load_records(progress_file)
    assessment_records = load_records(assessment_file)

    if not progress_records and not assessment_records:
        print("No student records are available.")
        return

    # Organize student progress by ID
    students = {}

    for record in progress_records:
        student_id = str(record.get("student_id", "")).strip()

        if student_id:
            students[student_id] = {
                "progress": record,
                "assessment_scores": []
            }

    # Add assessment results to the matching student
    for record in assessment_records:
        student_id = str(record.get("student_id", "")).strip()

        if not student_id:
            continue

        if student_id not in students:
            students[student_id] = {
                "progress": {},
                "assessment_scores": []
            }

        try:
            percentage = float(record.get("percentage", 0))
            students[student_id]["assessment_scores"].append(
                percentage
            )
        except (ValueError, TypeError):
            print("Warning: Invalid assessment score for", student_id)

    # Display combined reports
    print("\nTOTAL STUDENTS:", len(students))

    for student_id, data in students.items():

        progress = data["progress"]
        scores = data["assessment_scores"]

        print("\n--------------------------------")
        print("STUDENT ID:", student_id)

        if progress:
            completed = progress.get("completed_count", 0)
            total = progress.get("total_modules", 0)
            progress_percentage = progress.get(
                "progress_percentage", 0
            )

            print("Completed modules:", completed)
            print("Total modules:", total)
            print("Learning progress:", progress_percentage, "%")
        else:
            print("Learning progress: No progress record")

        if scores:
            average = sum(scores) / len(scores)

            print("Assessment attempts:", len(scores))
            print("Average assessment score:", round(average, 2), "%")
            print("Highest assessment score:", max(scores), "%")
        else:
            print("Assessment results: No assessment attempts")

    print("\n===== REPORT COMPLETED =====")

def cyberaware_dashboard():
    import json
    import os

    progress_file = "cyberaware_progress.json"
    assessment_file = "cyberaware_assessments.json"

    print("\n======================================")
    print("       CYBERAWARE DASHBOARD")
    print("======================================")

    # Read JSON files safely
    def load_records(filename):
        if not os.path.exists(filename):
            return []

        try:
            with open(filename, "r") as file:
                data = json.load(file)

            if isinstance(data, list):
                return data

            print("Warning: Invalid data format:", filename)
            return []

        except (json.JSONDecodeError, OSError):
            print("Warning: Could not read:", filename)
            return []

    progress_records = load_records(progress_file)
    assessment_records = load_records(assessment_file)

    # Store unique student IDs
    all_students = set()
    progress_students = {}
    assessment_students = set()

    # Count learning progress
    for record in progress_records:
        student_id = str(
            record.get("student_id", "")
        ).strip()

        if not student_id:
            continue

        all_students.add(student_id)

        # Keep the latest progress record for each student
        progress_students[student_id] = record

    # Count assessment attempts
    total_attempts = 0
    assessment_scores = []

    for record in assessment_records:
        student_id = str(
            record.get("student_id", "")
        ).strip()

        if student_id:
            all_students.add(student_id)
            assessment_students.add(student_id)

        try:
            percentage = float(record.get("percentage", 0))

            if 0 <= percentage <= 100:
                assessment_scores.append(percentage)
                total_attempts += 1

        except (ValueError, TypeError):
            continue

    # Calculate assessment average
    if assessment_scores:
        average_assessment = (
            sum(assessment_scores) / len(assessment_scores)
        )
    else:
        average_assessment = 0

    # Calculate learning progress
    progress_values = []

    for record in progress_students.values():
        try:
            percentage = float(
                record.get("progress_percentage", 0)
            )

            if 0 <= percentage <= 100:
                progress_values.append(percentage)

        except (ValueError, TypeError):
            continue

    if progress_values:
        average_progress = (
            sum(progress_values) / len(progress_values)
        )
    else:
        average_progress = 0

    # Display dashboard
    print("\nTOTAL STUDENTS:", len(all_students))

    print(
        "Students with progress records:",
        len(progress_students)
    )

    print(
        "Students with assessment records:",
        len(assessment_students)
    )

    print("Total assessment attempts:", total_attempts)

    print(
        "Average assessment score:",
        round(average_assessment, 2), "%"
    )

    print(
        "Average learning progress:",
        round(average_progress, 2), "%"
    )

    print("\n======================================")
    print("       DASHBOARD COMPLETED")
    print("======================================")

    assessment_file = "cyberaware_assessments.json"

    # Read JSON records safely
    def load_records(filename):
        if not os.path.exists(filename):
            return []

        try:
            with open(filename, "r") as file:
                data = json.load(file)

            if isinstance(data, list):
                return data

            print("Invalid data format:", filename)
            return []

        except (json.JSONDecodeError, OSError):
            print("Unable to read:", filename)
            return []

    progress_records = load_records(progress_file)
    assessment_records = load_records(assessment_file)

    # Find the student's progress record
    progress = {}

    for record in progress_records:
        saved_id = str(
            record.get("student_id", "")
        ).strip()

        if saved_id.lower() == student_id.lower():
            progress = record

    # Find the student's assessment records
    assessments = []

    for record in assessment_records:
        saved_id = str(
            record.get("student_id", "")
        ).strip()

        if saved_id.lower() == student_id.lower():
            assessments.append(record)

    # Check whether student exists
    if not progress and not assessments:
        print("No records found for this student.")
        return

    # Prepare PDF filename
    safe_id = re.sub(r"[^A-Za-z0-9_-]", "_", student_id)

    pdf_filename = "CyberAware_" + safe_id + "_Report.pdf"

    # Create PDF document
    document = SimpleDocTemplate(
        pdf_filename,
        pagesize=A4
    )

    styles = getSampleStyleSheet()
    content = []

    # PDF title
    content.append(
        Paragraph(
            "CYBERAWARE STUDENT PERFORMANCE REPORT",
            styles["Title"]
        )
    )

    content.append(Spacer(1, 0.2 * inch))

    content.append(
        Paragraph(
            "Anonymous Student ID: " + student_id,
            styles["Normal"]
        )
    )

    content.append(Spacer(1, 0.3 * inch))

    # Learning progress section
    content.append(
        Paragraph("LEARNING PROGRESS", styles["Heading2"])
    )

    if progress:
        completed = progress.get("completed_count", 0)
        total = progress.get("total_modules", 0)
        percentage = progress.get("progress_percentage", 0)

        progress_data = [
            ["Description", "Result"],
            ["Completed modules", str(completed)],
            ["Total modules", str(total)],
            ["Learning progress", str(percentage) + "%"]
        ]

        completed_modules = progress.get(
            "completed_modules", []
        )

        if isinstance(completed_modules, list):
            for module in completed_modules:
                content.append(
                    Paragraph(
                        "Completed: " + str(module),
                        styles["Normal"]
                    )
                )

    else:
        progress_data = [
            ["Description", "Result"],
            ["Learning progress", "No progress record"]
        ]

    progress_table = Table(progress_data, colWidths=[3 * inch, 2 * inch])

    progress_table.setStyle(
        TableStyle([
            ("BACKGROUND", (0, 0), (-1, 0), colors.lightblue),
            ("GRID", (0, 0), (-1, -1), 1, colors.grey),
            ("PADDING", (0, 0), (-1, -1), 8)
        ])
    )

    content.append(progress_table)
    content.append(Spacer(1, 0.3 * inch))

    # Assessment section
    content.append(
        Paragraph("ASSESSMENT PERFORMANCE", styles["Heading2"])
    )

    assessment_data = [
        ["Attempt", "Score", "Percentage"]
    ]

    valid_scores = []

    for number, record in enumerate(assessments, start=1):
        score = record.get("score", 0)
        total_questions = record.get("total_questions", 0)

        try:
            percentage = float(record.get("percentage", 0))

            assessment_data.append([
                str(number),
                str(score) + "/" + str(total_questions),
                str(percentage) + "%"
            ])

            valid_scores.append(percentage)

        except (ValueError, TypeError):
            continue

    if not valid_scores:
        assessment_data.append(
            ["-", "No assessment records", "-"]
        )

    assessment_table = Table(
        assessment_data,
        colWidths=[1.3 * inch, 1.7 * inch, 2 * inch]
    )

    assessment_table.setStyle(
        TableStyle([
            ("BACKGROUND", (0, 0), (-1, 0), colors.lightgreen),
            ("GRID", (0, 0), (-1, -1), 1, colors.grey),
            ("PADDING", (0, 0), (-1, -1), 8)
        ])
    )

    content.append(assessment_table)
    content.append(Spacer(1, 0.3 * inch))

    if valid_scores:
        average = sum(valid_scores) / len(valid_scores)

        content.append(
            Paragraph(
                "Total assessment attempts: " +
                str(len(valid_scores)),
                styles["Normal"]
            )
        )

        content.append(
            Paragraph(
                "Average assessment score: " +
                str(round(average, 2)) + "%",
                styles["Normal"]
            )
        )

    # Report footer
    content.append(Spacer(1, 0.4 * inch))

    content.append(
        Paragraph(
            "Generated by CyberAware Educational Application.",
            styles["Italic"]
        )
    )

    # Build and save PDF
    try:
        document.build(content)

        print("\nPDF report generated successfully!")
        print("File:", pdf_filename)

    except Exception as error:
        print("Could not generate PDF:", error)

def cyberaware_gui():
    import tkinter as tk
    from tkinter import messagebox
    import json
    import os

    # Read saved JSON files
    def load_records(filename):
        if not os.path.exists(filename):
            return []

        try:
            with open(filename, "r") as file:
                data = json.load(file)

            return data if isinstance(data, list) else []

        except (json.JSONDecodeError, OSError):
            return []

    # Create main window
    window = tk.Tk()
    window.title("CyberAware - Security Awareness")
    window.geometry("650x500")
    window.configure(bg="#eef3f8")

    # Application heading
    heading = tk.Label(
        window,

        fg="#123456"

    heading.pack(pady=15)

    subtitle = tk.Label(
        window,
        text="Cybersecurity Awareness Dashboard",
        font=("Arial", 12),
        bg="#eef3f8"
    )
    subtitle.pack(pady=5)

    # Dashboard information
    info_frame = tk.Frame(window, bg="#eef3f8")
    info_frame.pack(pady=20)

    student_label = tk.Label(
        info_frame,

        width=35,

    )
    student_label.pack(pady=5)

    attempt_label = tk.Label(

        font=("Arial", 13, "bold"),

        pady=12
    )
    attempt_label.pack(pady=5)

    average_label = tk.Label(
        info_frame,
        text="Average Assessment Score: 0%",
        font=("Arial", 13, "bold"),
        bg="white",
        width=35,
        pady=12
    )
    average_label.pack(pady=5)

    progress_label = tk.Label(
        info_frame,
        text="Average Learning Progress: 0%",
        font=("Arial", 13, "bold"),
        bg="white",
        width=35,
        pady=12
    )
    progress_label.pack(pady=5)

    # Refresh dashboard statistics
    def refresh_dashboard():
        progress_records = load_records(
            "cyberaware_progress.json"
        )

        assessment_records = load_records(
            "cyberaware_assessments.json"
        )

        student_ids = set()
        latest_progress = {}

        for record in progress_records:
            student_id = str(
                record.get("student_id", "")
            ).strip()

            if student_id:
                student_ids.add(student_id)
                latest_progress[student_id] = record

        scores = []

        for record in assessment_records:
            student_id = str(
                record.get("student_id", "")
            ).strip()

            if student_id:
                student_ids.add(student_id)

            try:
                score = float(record.get("percentage", 0))

                if 0 <= score <= 100:
                    scores.append(score)

            except (ValueError, TypeError):
                continue

        progress_values = []

        for record in latest_progress.values():
            try:
                value = float(
                    record.get("progress_percentage", 0)
                )

                if 0 <= value <= 100:
                    progress_values.append(value)

            except (ValueError, TypeError):
                continue

        average_score = (
            sum(scores) / len(scores) if scores else 0
        )

        average_progress = (
            sum(progress_values) / len(progress_values)
            if progress_values else 0
        )

        student_label.config(
            text="Total Students: " + str(len(student_ids))
        )

        attempt_label.config(
            text="Assessment Attempts: " + str(len(scores))
        )

        average_label.config(
            text="Average Assessment Score: "
            + str(round(average_score, 2)) + "%"
        )

        progress_label.config(
            text="Average Learning Progress: "
            + str(round(average_progress, 2)) + "%"
        )

            "cyberaware_assessments.json"
        )

            messagebox.showinfo(
                "Assessment Results",
                "No assessment records available."
            )
            return

        report = ""

        for record in records:
            report += (
                "\nStudent ID: "
                + str(record.get("student_id", "Unknown"))
                + "\nScore: "
                + str(record.get("score", 0))
                + "/"
                + str(record.get("total_questions", 0))
                + "\nPercentage: "
                + str(record.get("percentage", 0))
                + "%\n"
            )

        messagebox.showinfo(
            "Saved Assessment Results",
            report
        )

    button_frame = tk.Frame(window, bg="#eef3f8")
    button_frame.pack(pady=15)

    refresh_button = tk.Button(
        button_frame,
        text="Refresh Dashboard",
        command=refresh_dashboard,
        width=20,
        bg="#1769aa",
        fg="white"
    )
    refresh_button.pack(pady=5)

    results_button = tk.Button(
        button_frame,
        text="View Assessment Results",
        command=view_results,
        width=20,
        bg="#18865b",
        fg="white"
    )
    results_button.pack(pady=5)

    close_button = tk.Button(
        button_frame,
        text="Close Application",
        command=window.destroy,
        width=20,
        bg="#b83232",
        fg="white"
    )
    close_button.pack(pady=5)

    # Display current statistics
    refresh_dashboard()

    # Start GUI
    window.mainloop()
